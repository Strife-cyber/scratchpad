This example shows how to fill a registration form with various input types and handle validation.

<Tabs syncKey="lang">
  <TabItem label="Python">
    ```python
    import requests

    s = requests.post("http://localhost:8080/api/v1/sessions",
        json={"platform": "web"}).json()
    sid = s["sessionId"]
    base = f"http://localhost:8080/api/v1/sessions/{sid}/actions"

    def act(body):
        return requests.post(base, json=body).json()

    # Navigate to registration page
    act({"action": {"type": "navigate", "url": "https://example.com/register"}})
    act({"action": "wait", "condition": "network_idle"})

    # Fill form fields using fill_form (single action)
    act({
        "action": "fill_form",
        "form_fields": [
            {"selector": {"css": "#full-name"}, "value": "John Doe"},
            {"selector": {"css": "#email"}, "value": "john@example.com"},
            {"selector": {"css": "#password"}, "value": "securePass123"},
        ]
    })

    # Select from dropdown
    act({
        "action": "select_option",
        "selector": {"css": "#country"},
        "option_value": "US",
    })

    # Check a checkbox using the dedicated action
    act({"action": "check", "selector": {"css": "#agree-terms"}})

    # Submit form
    act({"action": "submit_form", "selector": {"css": "#registration-form"}})

    # Assert validation (if any)
    obs = act({
        "action": "assert",
        "assertion": {
            "type": "element_visible",
            "selector": {css: ".error-message"},
        },
    })
    if obs["assertion_result"]["success"]:
        print("Validation errors present")
    else:
        print("Form submitted successfully")

    requests.delete(f"http://localhost:8080/api/v1/sessions/{sid}")
    ```

  <TabItem label="TypeScript">
    ```typescript
    const { sessionId } = await fetch("http://localhost:8080/api/v1/sessions", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ platform: "web" }),
    }).then(r => r.json());

    const act = (body: any) =>
      fetch(`http://localhost:8080/api/v1/sessions/${sessionId}/actions`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
      }).then(r => r.json());

    await act({ action: { type: "navigate", url: "https://example.com/register" } });
    await act({ action: "wait", condition: "network_idle" });

    // Fill form fields in one call
    await act({
      action: "fill_form",
      form_fields: [
        { selector: { css: "#full-name" }, value: "John Doe" },
        { selector: { css: "#email" }, value: "john@example.com" },
        { selector: { css: "#password" }, value: "securePass123" },
      ],
    });
    await act({ action: "select_option", selector: { css: "#country" }, option_value: "US" });
    await act({ action: "check", selector: { css: "#agree-terms" } });
    await act({ action: "submit_form", selector: { css: "#registration-form" } });

    const obs = await act({
      action: "assert",
      assertion: { type: "element_visible", selector: { css: ".error-message" } },
    });
    console.log(obs.assertion_result.success ? "Errors found" : "Submitted OK");

    await fetch(`http://localhost:8080/api/v1/sessions/${sessionId}`, { method: "DELETE" });
    ```

## Key Takeaways

1. Use `fill_form` to fill multiple fields at once instead of individual `type` calls
2. Use `check`/`uncheck` instead of clicking checkboxes for clearer intent
3. Use `submit_form` to submit forms directly
4. Use `dismiss_modal` to handle cookie banners or popups that appear before forms
5. Assert after submission to check for validation errors or success
